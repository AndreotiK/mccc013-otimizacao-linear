var x1 >= 0;
var x2 >= 0;
var x3 >= 0;
var x4 >= 0;
var y1 >= 0;
var y2 >= 0;
var y3 >= 0;

minimize z: 20*(0.07)*x1 + 20*(0.06)*x2 
            + 20*(0.065)*x3 + 20*(0.075)*x4;

s.t. a: x1              - y1 = 2;
s.t. b: x2 + (1.060)*y1 - y2 = 4;
s.t. c: x3 + (1.055)*y2 - y3 = 8;
s.t. d: x4 + (1.045)*y3      = 5;

solve;
